function generateBlockingKeys(record: NormalizedRecord): string[] {

\section{Jaro-Winkler Similarity}
This snippet outlines the core function signature and algorithmic weight application for measuring text proximity.
\begin{lstlisting}[language=JavaScript]
function jaroWinkler(s1: string, s2: string): number {
  const jaro = jaroDistance(s1, s2);
  const prefixLen = commonPrefixLength(s1, s2, 4);
  return jaro + (prefixLen * 0.1 * (1 - jaro));
}
\end{lstlisting}

function sanitizeText(text: string): string {
  text = text.replace(/[\x00-\x08\x0B\x0C\x0E-\x1F\x7F]/g, '');
  text = text.substring(0, 500);
  text = text.replace(
    /ignore previous|system:|<\||\[INST\]/gi, '[REDACTED]'
  );
  return text;
}

\section{BFS Traversal}
This snippet details the bounded breadth-first search implementation used to rapidly explore node neighborhoods within the multi-tenant knowledge graph.
\begin{lstlisting}[language=JavaScript]
async function queryGraph(
  startRef: string, maxDepth = 2, limit = 100
): Promise<Subgraph> {
  const visited = new Set<string>();
  const queue: Array<[string, number]> = [[startRef, 0]];
  while (queue.length > 0) {
    const [nodeId, depth] = queue.shift()!;
    if (depth > maxDepth || visited.size >= limit) break;
    if (visited.has(nodeId)) continue;
    visited.add(nodeId);
    const neighbors = await getNeighbors(nodeId);
    for (const n of neighbors) {
      if (!visited.has(n.id)) queue.push([n.id, depth + 1]);
    }
  }
  return buildSubgraph(visited);
}
\end{lstlisting}
